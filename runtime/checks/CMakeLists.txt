cmake_minimum_required(VERSION 3.20)
project(PowerProsNativeChecks LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(DOLRECOMP_SOURCE "${PROJECT_ROOT}/local/vendor/DolRecomp/src")
set(DOL_CODE_LIBRARY "${PROJECT_ROOT}/local/build/powerpros-cpu/Release/powerpros_dol_code.lib" CACHE FILEPATH "Compiled DOL translation")
if(NOT EXISTS "${DOL_CODE_LIBRARY}")
    message(FATAL_ERROR "Compile the DOL translation before running native checks.")
endif()
add_executable(powerpros_native_checks native_memory.c "${DOLRECOMP_SOURCE}/cpu/cpu.c")
target_include_directories(powerpros_native_checks PRIVATE "${DOLRECOMP_SOURCE}")
target_link_libraries(powerpros_native_checks PRIVATE "${DOL_CODE_LIBRARY}")
file(SHA256 "${PROJECT_ROOT}/local/disc/DATA/sys/main.dol" DOL_SOURCE_SHA256)
target_compile_definitions(powerpros_native_checks PRIVATE DOL_SOURCE_SHA256="${DOL_SOURCE_SHA256}")
enable_testing()
add_test(NAME game_memory_routines COMMAND powerpros_native_checks "${PROJECT_ROOT}/local/analysis/memory-reference.bin")
