cmake_minimum_required(VERSION 3.20)
project(PowerProsCpuTranslationCheck LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(DOLRECOMP_SOURCE "${PROJECT_ROOT}/local/vendor/DolRecomp/src" CACHE PATH "DolRecomp source directory")
set(TRANSLATION_ROOT "${PROJECT_ROOT}/local/generated/dolrecomp" CACHE PATH "Generated local game code")

foreach(module IN ITEMS dol rel)
    if(module STREQUAL "dol")
        set(stem main)
    else()
        set(stem ovl_main)
    endif()
    set(manifest "${TRANSLATION_ROOT}/${module}/${stem}.c")
    if(NOT EXISTS "${manifest}")
        message(FATAL_ERROR "Missing generated ${module} code. Run the translation tools first.")
    endif()
    file(GLOB chunks CONFIGURE_DEPENDS "${TRANSLATION_ROOT}/${module}/${stem}_chunks/*.c")
    if(NOT chunks)
        message(FATAL_ERROR "Missing generated ${module} chunks.")
    endif()
    add_library(powerpros_${module}_code STATIC "${manifest}" ${chunks})
    target_include_directories(powerpros_${module}_code PRIVATE "${DOLRECOMP_SOURCE}")
    # First measure whether all output compiles; optimized/runtime builds come later.
    if(MSVC)
        target_compile_options(powerpros_${module}_code PRIVATE /Od /bigobj /wd4102)
    else()
        target_compile_options(powerpros_${module}_code PRIVATE -O0)
    endif()
endforeach()
